cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ttt2_launcher LANGUAGES CXX)
endif()
enable_language(RC)
add_executable(ttt2_launcher WIN32 launcher.cpp launcher.rc ../runtime/kernel/display_settings.cpp ../runtime/kernel/host_settings.cpp ../runtime/kernel/host_settings_windows.cpp)
target_compile_features(ttt2_launcher PRIVATE cxx_std_20)
target_compile_definitions(ttt2_launcher PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(ttt2_launcher PRIVATE comdlg32 shell32 comctl32 gdiplus uxtheme dwmapi ole32 dxgi)
set_target_properties(ttt2_launcher PROPERTIES OUTPUT_NAME "TTT2 Launcher")

add_executable(ttt2_preview WIN32 preview_main.cpp)
add_executable(ttt2_update_inplace WIN32 preview_main.cpp clean_updater.rc)
target_compile_features(ttt2_update_inplace PRIVATE cxx_std_20)
target_link_options(ttt2_update_inplace PRIVATE /MANIFEST:NO)
target_compile_definitions(ttt2_update_inplace PRIVATE UNICODE _UNICODE NOMINMAX TTT2_INPLACE_UPDATE=1)
target_link_libraries(ttt2_update_inplace PRIVATE comdlg32 shell32 bcrypt ole32 uuid comctl32)
set_target_properties(ttt2_update_inplace PROPERTIES OUTPUT_NAME "TTT2 Game Update" MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(TTT2_PREVIEW_PAYLOAD_DIR "" CACHE PATH "Optional staged offline preview payload for a standalone updater")
if(TTT2_PREVIEW_PAYLOAD_DIR)
    foreach(payload bin/game.exe bin/launcher.exe bin/SDL3.dll LICENSES/SDL-LICENSE.txt payload.sha256 README.txt build-info.json LICENSES/SMAA-LICENSE.txt LICENSES/SMAA-README.md)
        if(NOT EXISTS "${TTT2_PREVIEW_PAYLOAD_DIR}/${payload}")
            message(FATAL_ERROR "Standalone updater payload is missing ${payload}")
        endif()
    endforeach()
    file(TO_CMAKE_PATH "${TTT2_PREVIEW_PAYLOAD_DIR}" preview_payload_dir)
    set(preview_resources "${CMAKE_CURRENT_BINARY_DIR}/preview_payload.rc")
    file(WRITE "${preview_resources}" "101 RCDATA \"${preview_payload_dir}/bin/game.exe\"\n102 RCDATA \"${preview_payload_dir}/bin/launcher.exe\"\n103 RCDATA \"${preview_payload_dir}/payload.sha256\"\n104 RCDATA \"${preview_payload_dir}/README.txt\"\n105 RCDATA \"${preview_payload_dir}/build-info.json\"\n106 RCDATA \"${preview_payload_dir}/LICENSES/SMAA-LICENSE.txt\"\n107 RCDATA \"${preview_payload_dir}/LICENSES/SMAA-README.md\"\n")
    file(APPEND "${preview_resources}" "108 RCDATA \"${preview_payload_dir}/bin/SDL3.dll\"\n109 RCDATA \"${preview_payload_dir}/LICENSES/SDL-LICENSE.txt\"\n")
    set_property(SOURCE "${preview_resources}" PROPERTY OBJECT_DEPENDS
        "${preview_payload_dir}/bin/game.exe;${preview_payload_dir}/bin/launcher.exe;${preview_payload_dir}/bin/SDL3.dll;${preview_payload_dir}/LICENSES/SDL-LICENSE.txt;${preview_payload_dir}/payload.sha256;${preview_payload_dir}/README.txt;${preview_payload_dir}/build-info.json;${preview_payload_dir}/LICENSES/SMAA-LICENSE.txt;${preview_payload_dir}/LICENSES/SMAA-README.md")
    target_sources(ttt2_preview PRIVATE "${preview_resources}")
    target_compile_definitions(ttt2_preview PRIVATE TTT2_EMBEDDED_PREVIEW=1)
    target_sources(ttt2_update_inplace PRIVATE "${preview_resources}")
    target_compile_definitions(ttt2_update_inplace PRIVATE TTT2_EMBEDDED_PREVIEW=1)
endif()
target_compile_features(ttt2_preview PRIVATE cxx_std_20)
target_compile_definitions(ttt2_preview PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(ttt2_preview PRIVATE comdlg32 shell32 bcrypt ole32 uuid)
set_target_properties(ttt2_preview PROPERTIES OUTPUT_NAME "TTT2 Offline Update" MSVC_RUNTIME_LIBRARY "MultiThreaded")
